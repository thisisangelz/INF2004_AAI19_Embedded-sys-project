cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w CACHE STRING "Board type")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
include(../pico_rssi_tracker/pico_sdk_import.cmake)
project(tcs34725_diag C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    pico_sdk_init()
endif()
add_executable(tcs34725_diag tcs34725_diag.c)
target_compile_options(tcs34725_diag PRIVATE -Wall -Wextra)
target_link_libraries(tcs34725_diag pico_stdlib hardware_i2c)
pico_enable_stdio_usb(tcs34725_diag 1)
pico_enable_stdio_uart(tcs34725_diag 0)
pico_add_extra_outputs(tcs34725_diag)
